let commit = "", language = "";

if (gh) {
  try {
    const api = `https://api.github.com/repos/${gh[1]}/${gh[2]}`;
    const info = (await tp.obsidian.requestUrl({ url: api })).json;
    language = info.language ?? "";
    const commits = (await tp.obsidian.requestUrl({ url: `${api}/commits?sha=${info.default_branch}&per_page=1` })).json;
    commit = commits[0]?.sha ?? "";
  } catch (e) {
    console.warn("GitHub API 讀取失敗", e);
  }
}
if (!commit) commit = (await tp.system.prompt("閱讀時的 commit hash（GitHub 頁面上的最新 commit）")) ?? "";
